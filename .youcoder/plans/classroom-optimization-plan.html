import { apiClient } from '../../api/apiClient';

export interface TTSResponse {
  audioUrl: string;
  durationMs: number;
}

export async function synthesize(text: string): Promise<TTSResponse> {
  // 方案 A: 后端 API 合成
  try {
    const res = await apiClient.post<TTSResponse>('/api/tts/synthesize', { text });
    return res.data;
  } catch {
    // 方案 B: 浏览器原生 TTS fallback
    return synthesizeBrowser(text);
  }
}

function synthesizeBrowser(text: string): Promise<TTSResponse> {
  return new Promise((resolve, reject) => {
    if (!window.speechSynthesis) {
      reject(new Error('TTS not supported'));
      return;
    }
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = 'zh-CN';
    utterance.rate = 1.0;
    // 用 blob URL 模拟（实际用 AudioContext 录制—简化版直接用时长估算）
    utterance.onend = () => resolve({
      audioUrl: '',  // 浏览器原生不需要 URL
      durationMs: (text.length / 4) * 1000,  // 粗略估算
    });
    utterance.onerror = reject;
    window.speechSynthesis.speak(utterance);
  });
}

验收：API 调用成功后返回音频 URL，可被 Audio 元素播放

const autoPlayAction = useCallback(() => {
  if (currentAction.type === 'speech') {

  } else if (currentAction.type === 'scene_transition') {

    actionTimerRef.current = setTimeout(nextAction, 1500);
  }
  // quiz/discussion 等待用户交互
}, [currentAction, nextAction, ttsPlayer]);

const handleMouseDown = (e: React.MouseEvent) => {
  e.preventDefault();
  const startX = e.clientX;
  const startWidth = sidebarWidth;
  const handleMove = (me: MouseEvent) => {
    const newWidth = Math.min(MAX_WIDTH, Math.max(MIN_WIDTH, startWidth + me.clientX - startX));
    setSidebarWidth(newWidth);
  };
  const handleUp = () => {
    document.removeEventListener('mousemove', handleMove);
    document.removeEventListener('mouseup', handleUp);
  };
  document.addEventListener('mousemove', handleMove);
  document.addEventListener('mouseup', handleUp);
};

import ReactMarkdown from 'react-markdown';
import rehypeKatex from 'rehype-katex';
import remarkMath from 'remark-math';
import { Prism as SyntaxHighlighter } from 'react-syntax-highlighter';
import { oneDark } from 'react-syntax-highlighter/dist/esm/styles/prism';

// 渲染 Markdown 内容
<ReactMarkdown
  remarkPlugins={[remarkMath]}
  rehypePlugins={[rehypeKatex]}
  components={{
    code({ node, inline, className, children, ...props }) {
      const match = /language-(\w+)/.exec(className || '');
      return !inline && match
        ? <SyntaxHighlighter style={oneDark} language={match[1]} PreTag="div" {...props}>
            {String(children).replace(/\n$/, '')}
          </SyntaxHighlighter>
        : <code className={className} {...props}>{children}</code>;
    },
    img({ src, alt }) {
      return <img src={src} alt={alt} style={{ maxWidth: '100%', borderRadius: 8 }} />;
    },
  }}
>
  {content}
</ReactMarkdown>

验收：白板内容中支持 Markdown 格式、代码高亮、LaTeX 公式和图片
